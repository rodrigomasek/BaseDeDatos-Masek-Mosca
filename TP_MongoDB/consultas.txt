// ---------------------------------------------------------------------
// 3. Agregar un campo "ultima_actualizacion" con la fecha actual a todos
//    los documentos de restaurantes. Siempre que se actualice el
//    documento se debe setear ese valor con la fecha actual.
// ---------------------------------------------------------------------
// $currentDate pone la fecha y hora actual del servidor. Por eso, en
// TODAS las consultas que modifican un restaurante (la 6 y la 8) se
// vuelve a usar $currentDate sobre este mismo campo.
db.restaurantes.updateMany(
  {},
  { $currentDate: { ultima_actualizacion: true } }
)


// ---------------------------------------------------------------------
// 4. Mostrar el nombre de los restaurantes que cuentan con alguna
//    promoción del 10%.
// ---------------------------------------------------------------------
db.restaurantes.find(
  { "promociones.porcentaje": 10 },
  { _id: 0, nombre: 1 }
)


// ---------------------------------------------------------------------
// 5. Mostrar el promedio de calificación de las reseñas de cada
//    restaurante.
// ---------------------------------------------------------------------
db.restaurantes.aggregate([
  { $project: {
      _id: 0,
      nombre: 1,
      promedio_calificacion: { $round: [ { $avg: "$resenas.calificacion" }, 2 ] }
  } }
])


// ---------------------------------------------------------------------
// 6. Modificar el horario de apertura a las 11:30hs para los
//    restaurantes que tienen delivery o reservas online.
// ---------------------------------------------------------------------
db.restaurantes.updateMany(
  { servicios: { $in: ["Delivery", "Reservas online"] } },
  {
    $set: { "informacion.horario.apertura": "11:30" },
    $currentDate: { ultima_actualizacion: true }
  }
)

// Control: cómo quedó el horario de apertura de cada uno
db.restaurantes.find(
  {},
  { _id: 0, nombre: 1, "informacion.horario.apertura": 1 }
)


// ---------------------------------------------------------------------
// 7. Mostrar el nombre, dirección y teléfono de los restaurantes que
//    tengan como evento especial "Música en vivo".
// ---------------------------------------------------------------------
db.restaurantes.find(
  { "eventos.nombre": "Música en vivo" },
  { _id: 0, nombre: 1, "informacion.direccion": 1, "informacion.telefono": 1 }
)


// ---------------------------------------------------------------------
// 8. Agregar una promoción que inicie el 2026-09-10 y termine el
//    2026-10-10. Elegir un restaurante al azar.
// ---------------------------------------------------------------------
// $sample elige un documento al azar de la colección.
var elegido = db.restaurantes.aggregate([ { $sample: { size: 1 } } ]).toArray()[0]

db.restaurantes.updateOne(
  { _id: elegido._id },
  {
    $push: { promociones: {
      nombre: "Promo primavera",
      porcentaje: 20,
      fecha_inicio: ISODate("2026-09-10"),
      fecha_fin: ISODate("2026-10-10")
    } },
    $currentDate: { ultima_actualizacion: true }
  }
)

// Control: a qué restaurante le tocó y cómo quedaron sus promociones
db.restaurantes.find(
  { _id: elegido._id },
  { _id: 0, nombre: 1, promociones: 1 }
)


// ---------------------------------------------------------------------
// 9. Mostrar el nombre del restaurante, sus servicios adicionales y menú
//    para aquel que posea el máximo descuento para el día de hoy.
// ---------------------------------------------------------------------
// Una promoción vale "hoy" si ya empezó y todavía no terminó.
db.restaurantes.aggregate([
  { $unwind: "$promociones" },
  { $match: {
      "promociones.fecha_inicio": { $lte: new Date() },
      "promociones.fecha_fin":    { $gte: new Date() }
  } },
  { $sort: { "promociones.porcentaje": -1 } },
  { $limit: 1 },
  { $project: {
      _id: 0,
      nombre: 1,
      descuento: "$promociones",
      servicios: 1,
      menu: 1
  } }
])


// ---------------------------------------------------------------------
// 10. Listar el nombre de los 3 restaurantes que tienen la mejor
//     calificación promedio.
// ---------------------------------------------------------------------
db.restaurantes.aggregate([
  { $project: {
      _id: 0,
      nombre: 1,
      promedio_calificacion: { $round: [ { $avg: "$resenas.calificacion" }, 2 ] }
  } },
  { $sort: { promedio_calificacion: -1 } },
  { $limit: 3 }
])


// ---------------------------------------------------------------------
// 11. Mostrar los restaurantes que tengan descuento para jubilados o
//     para estudiantes.
// ---------------------------------------------------------------------
// La "i" de la expresión regular hace que no importen las mayúsculas.
db.restaurantes.find(
  { "promociones.nombre": { $regex: /jubilad|estudiant/i } },
  { _id: 0, nombre: 1, "promociones.nombre": 1, "promociones.porcentaje": 1 }
)


// ---------------------------------------------------------------------
// 12. Mostrar los restaurantes que tengan delivery, reservas online y
//     además alguna promoción vigente mayor al 20%.
// ---------------------------------------------------------------------
// $all exige los dos servicios a la vez.
// $elemMatch exige que UNA MISMA promoción sea mayor al 20% Y esté
// vigente. Sin $elemMatch, un restaurante con una promo vencida del 25%
// y otra vigente del 10% aparecería por error.
db.restaurantes.find(
  {
    servicios: { $all: ["Delivery", "Reservas online"] },
    promociones: { $elemMatch: {
      porcentaje:   { $gt: 20 },
      fecha_inicio: { $lte: new Date() },
      fecha_fin:    { $gte: new Date() }
    } }
  },
  { _id: 0, nombre: 1, servicios: 1, promociones: 1 }
)


// ---------------------------------------------------------------------
// 13. Mostrar los 3 servicios adicionales más comunes ofrecidos por los
//     restaurantes en general.
// ---------------------------------------------------------------------
db.restaurantes.aggregate([
  { $unwind: "$servicios" },
  { $group: { _id: "$servicios", cantidad_restaurantes: { $sum: 1 } } },
  { $sort: { cantidad_restaurantes: -1 } },
  { $limit: 3 }
])


// ---------------------------------------------------------------------
// 14. Mostrar todos los restaurantes (nombre, dirección y teléfono) que
//     no tengan promociones.
// ---------------------------------------------------------------------
// "No tener promociones" son dos casos: que el campo no exista, o que
// exista pero la lista esté vacía.
db.restaurantes.find(
  { $or: [
      { promociones: { $exists: false } },
      { promociones: { $size: 0 } }
  ] },
  { _id: 0, nombre: 1, "informacion.direccion": 1, "informacion.telefono": 1 }
)


// ---------------------------------------------------------------------
// 15. Mostrar todos los restaurantes (nombre, dirección y teléfono) que
//     tienen, al menos, tres promociones.
// ---------------------------------------------------------------------
// "promociones.2" es la TERCERA promoción (la lista empieza en 0): si
// existe, el restaurante tiene 3 o más.
db.restaurantes.find(
  { "promociones.2": { $exists: true } },
  { _id: 0, nombre: 1, "informacion.direccion": 1, "informacion.telefono": 1 }
)


// ---------------------------------------------------------------------
// 16. Mostrar todos los restaurantes que tengan en su menú un plato
//     "sin TACC".
// ---------------------------------------------------------------------
db.restaurantes.find(
  { "menu.etiquetas": "sin TACC" },
  { _id: 0, nombre: 1 }
)


// ---------------------------------------------------------------------
// 17. Mostrar los restaurantes con sus promociones vigentes al día de
//     hoy.
// ---------------------------------------------------------------------
// $filter se queda solo con las promociones vigentes de cada
// restaurante ($$NOW es la fecha y hora actual). Después se descartan
// los restaurantes que no tienen ninguna vigente.
db.restaurantes.aggregate([
  { $project: {
      _id: 0,
      nombre: 1,
      promociones_vigentes: { $filter: {
        input: "$promociones",
        as: "promo",
        cond: { $and: [
          { $lte: [ "$$promo.fecha_inicio", "$$NOW" ] },
          { $gte: [ "$$promo.fecha_fin", "$$NOW" ] }
        ] }
      } }
  } },
  { $match: { "promociones_vigentes.0": { $exists: true } } }
])


// ---------------------------------------------------------------------
// 18. Mostrar el evento más próximo a realizarse de cada restaurante.
// ---------------------------------------------------------------------
// Se queda con los eventos que todavía no pasaron, los ordena por fecha
// y toma el primero de cada restaurante.
db.restaurantes.aggregate([
  { $unwind: "$eventos" },
  { $match: { "eventos.fecha": { $gte: new Date() } } },
  { $sort: { "eventos.fecha": 1 } },
  { $group: {
      _id: "$_id",
      restaurante: { $first: "$nombre" },
      proximo_evento: { $first: "$eventos" }
  } },
  { $sort: { "proximo_evento.fecha": 1 } },
  { $project: { _id: 0, restaurante: 1, proximo_evento: 1 } }
])


// ---------------------------------------------------------------------
// 19. Borrar todos los documentos de la colección de restaurantes que
//     no tengan menú. Tener en cuenta tanto los casos en los que existe
//     como en los que no existe el campo.
// ---------------------------------------------------------------------
// Control previo: cuáles se van a borrar
db.restaurantes.find(
  { $or: [
      { menu: { $exists: false } },
      { menu: { $size: 0 } }
  ] },
  { _id: 0, nombre: 1 }
)

db.restaurantes.deleteMany(
  { $or: [
      { menu: { $exists: false } },
      { menu: { $size: 0 } }
  ] }
)


// ---------------------------------------------------------------------
// 20. Mostrar el precio promedio, el plato más caro y el plato más
//     barato del menú de cada restaurante.
// ---------------------------------------------------------------------
// Se ordenan los platos de cada restaurante de más barato a más caro:
// el primero ($first) es el más barato y el último ($last) el más caro.
db.restaurantes.aggregate([
  { $unwind: "$menu" },
  { $sort: { "menu.precio": 1 } },
  { $group: {
      _id: "$_id",
      restaurante: { $first: "$nombre" },
      precio_promedio: { $avg: "$menu.precio" },
      plato_mas_barato: { $first: { nombre: "$menu.nombre", precio: "$menu.precio" } },
      plato_mas_caro: { $last: { nombre: "$menu.nombre", precio: "$menu.precio" } }
  } },
  { $project: {
      _id: 0,
      restaurante: 1,
      precio_promedio: { $round: [ "$precio_promedio", 2 ] },
      plato_mas_barato: 1,
      plato_mas_caro: 1
  } },
  { $sort: { restaurante: 1 } }
])
